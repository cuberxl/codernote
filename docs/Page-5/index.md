### 常用数学函数

```cpp
int abs(int x);             // 求x的绝对值
double fabs(double x);      // 求x的绝对值
double exp(double x);       // 求x的指数
double log(double x);       // 求x的自然对数
double log10(double x);     // 求x的以10为底的对数
double log2(double x);      // 求x的以2为底的对数
double sqrt(double x);      // 求x的平方根
double cbrt(double x);      // 求x的立方根
double pow(double x, double y); // 求x的y次幂
double ceil(double x);      // 将x向上取整
double floor(double x);     // 将x向下取整
double sin(double x);       // 求x的正弦值
double cos(double x);       // 求x的余弦值
double tan(double x);       // 求x的正切值
double asin(double x);      // 求x的反正弦值
double acos(double x);      // 求x的反余弦值
double atan(double x);      // 求x的反正切值
double atan2(double y, double x); // 求y/x的反正切值
double sinh(double x);      // 求x的双曲正弦值
double cosh(double x);      // 求x的双曲余弦值
double tanh(double x);      // 求x的双曲正切值
double asinh(double x);     // 求x的双曲反正弦值
double acosh(double x);     // 求x的双曲反余弦值
double atanh(double x);     // 求x的双曲反正切值
```
